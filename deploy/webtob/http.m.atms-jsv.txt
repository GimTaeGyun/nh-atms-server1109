# WebtoB(nbatmsltwb01, 5.0.0.4) http.m 추가 블록 — NH-ATMS JEUS 매니지드 서버 6개 연동 (2026-10-26 구성)
#   대상 파일: /WEBTOB/webtobATMS/config/http.m (webatms 소유, WebtoB 담당 또는 root 작업일 ROOT_DAY 2-13)
#   이 파일은 "붙여 넣을 내용" 이다 — 파일 통째로 교체하지 않는다
#
#   1) *SERVER 이름 = JEUS 쪽 WebtoB 커넥터의 registration-id (같아야 연결됨)
#      아래는 서버 이름과 같다고 가정. WAS 담당이 다른 id 를 정했으면 *SERVER 와 *URI 의 SvrName 을 함께 바꾼다
#   2) *NODE 에 JSVPORT = 19000 이 있어야 함 (2026-10-15 :19000 LISTEN 확인 → 있을 가능성 높음)
#   3) 이미 JSV 서버그룹이 있으면 *SVRGROUP 은 넣지 않고 그 그룹 이름을 SvgName 에 (*NODE 는 기존 항목 확인용, 통째로 붙이지 않음)
#   4) *URI 는 구체적인 경로를 위에 (/api/journal/ 이 /api/ 보다 먼저). DOCROOT·기존 URI·오류 페이지는 그대로
#   5) MinProc/MaxProc 은 예시 — 담당 기준
#   6) 2026-11-09: 모듈 간 호출도 이 9000 을 거친다 (소스 jeus-common.yml jeus.webtob-url = http://16.88.122.112:9000)
#      web → /api/file/remote/ (file01), file → /api/remote-exec/ (atms01), web 중계 → /api/agent/legacy/ (ajou01),
#      journal → /api/if/internal/·/api/if/external/ (ifin01·ifex01)
#      → 아래 *URI 그대로 대상이 맞으므로 추가할 줄은 없다. 단 ws01(16.88.122.113) → wb01:9000 이 열려 있어야 함 [확인 2026-10-15]
#      /atms/·/atmr/·/atmf/ 규칙은 에이전트 옛 경로 호환용. 모듈 간 호출이 /atms/·/atmr/ 를 쓰면 atms01 로 되돌아와 루프
#   적용: (webatms) wscfl -i http.m → wsdown -i → wsboot
#   확인: ss -tn | grep 19000 에 16.88.122.113 ESTAB (서버당 연결 수는 커넥터 설정에 따라 여러 줄),
#         모듈 경로(/api/journal/ 등)가 302·503 이 아니면 그 서버까지 연결,
#         curl -s -o /dev/null -w "%{http_code}\n" http://127.0.0.1:9000/v3/api-docs → 200

# --- *NODE : 기존 노드 항목에 아래 값이 있는지만 확인 (없는 항목만 추가, 나머지 기존 값 유지)
#     노드 이름은 hostname 과 같아야 함 (nbatmsltwb01)
*NODE
nbatmsltwb01    WEBTOBDIR = "/WEBTOB/webtobATMS",      # 기존 값 유지
                DOCROOT   = "/home/atmswas/docs",       # 확인된 DocBase — 바꾸지 않음 (배포가 이 폴더를 통째로 교체)
                PORT      = "9000",                     # 기존 값 유지
                JSVPORT   = 19000                       # JEUS 커넥터 6개가 접속 (없으면 추가)
                # ... 그 밖의 기존 항목(SHMKEY, HTH, LOGGING, ERRORDOCUMENT 등)은 그대로
